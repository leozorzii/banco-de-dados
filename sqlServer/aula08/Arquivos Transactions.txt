--questao feita com um procedure
	CREATE PROCEDURE usp_transferencia
	@origem INT,
	@destino INT,
	@valor DECIMAL(10,2)
	AS
		BEGIN
			BEGIN TRANSACTION
			IF((SELECT  saldo FROM CONTA WHERE id = @origem) >= @valor)
				BEGIN
					UPDATE CONTA
					SET saldo = saldo - @valor
					WHERE id = @origem


					UPDATE CONTA
					SET saldo = saldo + @valor
					WHERE ID = @destino
					COMMIT TRAN
					PRINT 'Saldo atualizado com sucesso'
				END

			ELSE 
				BEGIN
					PRINT 'Saldo insuficiente'
					ROLLBACK TRANSACTION
				END
	END
GO
